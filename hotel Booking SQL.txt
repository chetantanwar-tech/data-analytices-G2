SELECT *
FROM hotel_bookings
LIMIT 10;

--Most Preferred Meal Type--
SELECT meal,
       COUNT(*) AS preference_count
FROM hotel_bookings
GROUP BY meal
ORDER BY preference_count DESC;

--ADR Analysis by Customer Type--
SELECT customer_type,
       ROUND(AVG(adr)::numeric, 2) AS avg_adr
FROM hotel_bookings
GROUP BY customer_type
ORDER BY avg_adr DESC;

--Repeated Guests Analysis--
SELECT is_repeated_guest,
       COUNT(*) AS total_guests
FROM hotel_bookings
GROUP BY is_repeated_guest;

--Reservation Status Distribution--
SELECT reservation_status,
       COUNT(*) AS total_reservations
FROM hotel_bookings
GROUP BY reservation_status
ORDER BY total_reservations DESC;

--Parking Space Requirement Analysis--
SELECT required_car_parking_spaces,
       COUNT(*) AS total_bookings
FROM hotel_bookings
GROUP BY required_car_parking_spaces
ORDER BY total_bookings DESC;

--Window Function Query--
SELECT hotel,
       arrival_date_year,
       COUNT(*) AS total_bookings,
       RANK() OVER (
           ORDER BY COUNT(*) DESC
       ) AS booking_rank
FROM hotel_bookings
GROUP BY hotel, arrival_date_year;
